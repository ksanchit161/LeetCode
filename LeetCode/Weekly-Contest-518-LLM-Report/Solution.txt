1class Solution {
2    public int countRotations(String s, int k) {
3        int n = s.length();
4        boolean[] eq = new boolean[n];
5        int totalEq = 0;
6
7        for (int i = 0; i < n; i++) {
8            eq[i] = s.charAt(i) == s.charAt((i + 1) % n);
9            if (eq[i]) totalEq++;
10        }
11
12        int ans = 0;
13        for (int prefix = 0; prefix < n; prefix++) {
14            int missingEdge = (prefix - 1 + n) % n;
15            int score = totalEq - (eq[missingEdge] ? 1 : 0);
16            if (score == k) ans++;
17        }
18
19        return ans;
20    }
