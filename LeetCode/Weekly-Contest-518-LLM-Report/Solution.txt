1#include <string>
2using namespace std;
3
4class Solution {
5public:
6    int countRotations(string s, int k) {
7        int n = (int)s.size();
8        int equalEdges = 0;
9
10        for (int i = 0; i < n; ++i) {
11            if (s[i] == s[(i + 1) % n]) {
12                ++equalEdges;
13            }
14        }
15
16        if (k == equalEdges - 1) {
17            return equalEdges;
18        }
19        if (k == equalEdges) {
20            return n - equalEdges;
21        }
22        return 0;
23    }
24};
