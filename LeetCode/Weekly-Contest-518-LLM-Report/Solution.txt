1#include <string>
2using namespace std;
3
4class Solution {
5public:
6    int countRotations(string s, int k) {
7        int n = s.size();
8        int equalPairs = 0;
9
10        for (int i = 0; i < n; ++i) {
11            if (s[i] == s[(i + 1) % n]) {
12                ++equalPairs;
13            }
14        }
15
16        if (k == equalPairs) {
17            return n - equalPairs;
18        }
19        if (k == equalPairs - 1) {
20            return equalPairs;
21        }
22
23        return 0;
24    }
25};
